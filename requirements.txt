# GPU-accelerated hand tracking (Settings -> Camera -> "GPU Mode").
# onnxruntime-directml ships the DmlExecutionProvider, which uses
# DirectX 12 for inference — works on any modern NVIDIA / AMD /
# Intel GPU on Windows 10+. The corresponding ONNX models live in
# assets/models/ (palm_detector.onnx + hand_landmark.onnx + the
# 2016-row anchor table palm_anchors.npy). Same model weights as
# the default MediaPipe CPU path so accuracy is identical; the
# whole point of the swap is the ~25x palm-detect speedup
# (15-25 ms CPU -> 0.8 ms GPU). The runtime falls back to CPU
# MediaPipe transparently if the DML provider isn't reachable on
# a particular machine, so this stays an opt-in toggle.
#
# IMPORTANT: do NOT also have `onnxruntime` (the CPU-only wheel)
# installed alongside this — they share the same `onnxruntime`
# import name and will silently shadow each other. Pick one.
onnxruntime-directml

# Optional — only needed by the experimental Live API agent
# (src/hgr/live_api). The rest of the app launches fine without it;
# clicking "Test Live API" in the UI shows a friendly install hint.
websocket-client

# OCR for the Live API agent's click_text_on_screen tool (find a button by
# its visible text and click its exact pixels — for Chromium/webview UIs
# UIA can't see, e.g. Valorant's PLAY). Pure ONNX, offline.
#
# CRITICAL: rapidocr-onnxruntime DEPENDS ON the CPU-only `onnxruntime`
# wheel, which shadows onnxruntime-directml above and KILLS the
# DmlExecutionProvider (GPU mode). Install it WITHOUT pulling that wheel:
#   pip install --no-deps rapidocr-onnxruntime
#   pip install pyclipper shapely pyyaml   # rapidocr's other runtime deps
# (numpy / opencv / pillow are already required above.) If a plain
# `pip install -r requirements.txt` ever re-adds CPU onnxruntime, restore
# DirectML with:
#   pip uninstall -y onnxruntime && pip install --force-reinstall --no-deps onnxruntime-directml
rapidocr-onnxruntime
pyclipper
shapely
pyyaml

# Phone camera over WebRTC (Settings -> Camera -> "Connect Phone", the
# pairing-code flow via touchless-control.com/connect). The WebRTC stream
# is decoded by a hidden Chromium page (PySide6 QtWebEngine — already a
# PySide6 component, no extra wheel), which feeds JPEG frames to a tiny
# local aiohttp server. aiohttp is also used by the QR server; pinned
# here explicitly. (No aiortc — QtWebEngine does the WebRTC + hardware
# decode, which is far smoother than software decoding in Python.)
aiohttp

# Optional — Google connectors for the Live API agent (Gmail / Calendar /
# Docs, under src/hgr/live_api/connectors). Pure-Python, no numpy/onnxruntime

# ---- GPU / ML inference ----------------------------------------------------
# Windows uses onnxruntime-directml (DirectX 12). macOS has NO DirectML; use the
# plain onnxruntime wheel, which ships the CoreMLExecutionProvider (Apple Neural
# Engine / GPU) plus a CPU fallback. Same ONNX model files in assets/models/.
# IMPORTANT: never install onnxruntime-directml on macOS, and never have BOTH
# `onnxruntime` and `onnxruntime-directml` installed (they share the import name
# and silently shadow each other).
onnxruntime

pyobjc-framework-CoreAudio
